```sql
  display_name text not null,
  avatar_url   text,
  avatar_color_index integer not null default 0 check (avatar_color_index between 0 and 7),
  timezone     text not null default 'UTC',
  push_token   text,
  created_at   timestamptz not null default now()
);

-- Houses
create table public.houses (
  id         uuid primary key default uuid_generate_v4(),
  name       text not null,
  created_by uuid not null references public.profiles(id),
  created_at timestamptz not null default now(),
  code       text not null unique
);
create index on public.houses(code);

-- House members (one house per user enforced by UNIQUE)
create table public.house_members (
  id        uuid primary key default uuid_generate_v4(),
  house_id  uuid not null references public.houses(id) on delete cascade,
  user_id   uuid not null references public.profiles(id) on delete cascade,
  joined_at timestamptz not null default now(),
  unique(user_id)  -- Phase 1: one house per user
);
create index on public.house_members(house_id);

-- Chores
create type public.chore_frequency as enum ('daily', 'weekly');

create table public.chores (
  id          uuid primary key default uuid_generate_v4(),
  house_id    uuid not null references public.houses(id) on delete cascade,
  created_by  uuid not null references public.profiles(id),
  name        text not null,
  assigned_to uuid references public.profiles(id) on delete set null,
  due_date    date,
  notes       text,

  created_at  timestamptz not null default now()
);

create index on public.chores(assigned_to);

create table public.chore_completions (
  id           uuid primary key default uuid_generate_v4(),
  chore_id     uuid not null references public.chores(id) on delete cascade,
  completed_by uuid not null references public.profiles(id),
  completed_at timestamptz not null default now(),
  for_date     date not null,
```

- [ ] **Create `supabase/migrations/20260623000002_rls.sql`**

```sql
-- Enable RLS on all tables
alter table public.profiles enable row level security;
alter table public.houses enable row level security;
alter table public.house_members enable row level security;
alter table public.chores enable row level security;
alter table public.chore_completions enable row level security;

-- Helper: get current user's house_id
create or replace function public.my_house_id()
returns uuid language sql stable security definer as $$
  select house_id from public.house_members where user_id = auth.uid()
$$;

-- Profiles: users can read all profiles in their house; update only their own
create policy "profiles: read housemates"
  on public.profiles for select
  using (
    id = auth.uid() or
    id in (
      select user_id from public.house_members
      where house_id = public.my_house_id()
    )
  );

create policy "profiles: insert own"
  on public.profiles for insert
  with check (id = auth.uid());

create policy "profiles: update own"
  on public.profiles for update
  using (id = auth.uid());

-- Houses: members can read their house; any authed user can insert
create policy "houses: read own"
  on public.houses for select
  using (id = public.my_house_id());

create policy "houses: insert"
  on public.houses for insert
  with check (auth.uid() is not null);

-- House members: read all members of own house; insert only self; delete only self
create policy "house_members: read housemates"
  on public.house_members for select
  using (house_id = public.my_house_id() or user_id = auth.uid());

create policy "house_members: join"
  on public.house_members for insert
  with check (user_id = auth.uid());

create policy "house_members: leave"
  on public.house_members for delete
  using (user_id = auth.uid());

-- Chores: full CRUD for house members
create policy "chores: house members"
  on public.chores for all
  using (house_id = public.my_house_id())
  with check (house_id = public.my_house_id());

-- Completions: house members can read; users can only insert/delete their own
create policy "completions: read"
  on public.chore_completions for select
  using (
    chore_id in (
      select id from public.chores where house_id = public.my_house_id()
    )
  );

create policy "completions: insert own"
  on public.chore_completions for insert
  with check (completed_by = auth.uid());

create policy "completions: delete own"
  on public.chore_completions for delete
  using (completed_by = auth.uid());
```

```typescript
import { useEffect, useState } from 'react'
import { Session } from '@supabase/supabase-js'
import { supabase } from '@/lib/supabase'
import { Profile } from '@/types/database'

interface AuthState {
  session: Session | null
  profile: Profile | null
  loading: boolean
}

export function useAuth() {
  const [state, setState] = useState<AuthState>({
    session: null,
    profile: null,
    loading: true,
  })

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      if (session) loadProfile(session)
      else setState({ session: null, profile: null, loading: false })
    })

    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      async (_event, session) => {
        if (session) await loadProfile(session)
        else setState({ session: null, profile: null, loading: false })
      }
    )
    return () => subscription.unsubscribe()
  }, [])

  async function loadProfile(session: Session) {
    const { data } = await supabase
      .from('profiles')
      .select('*')
      .eq('id', session.user.id)
      .single()
    setState({ session, profile: data, loading: false })
  }

  return state
}
```

```typescript
  }, [session, profile, loading])

      <Slot />

  )
```

```typescript
import { useState } from 'react'
import { View, Text, TextInput, TouchableOpacity, StyleSheet, Platform, Alert } from 'react-native'
import { LinearGradient } from 'expo-linear-gradient'
import * as AppleAuthentication from 'expo-apple-authentication'
import { supabase } from '@/lib/supabase'
import { Colors, PrimaryGradient, Radii } from '@/constants/theme'

export default function SignInScreen() {
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [isSignUp, setIsSignUp] = useState(false)
  const [loading, setLoading] = useState(false)

  async function handleApple() {
    try {
      const credential = await AppleAuthentication.signInAsync({
        requestedScopes: [
          AppleAuthentication.AppleAuthenticationScope.FULL_NAME,
          AppleAuthentication.AppleAuthenticationScope.EMAIL,
        ],
      })
      const { error } = await supabase.auth.signInWithIdToken({
        provider: 'apple',
        token: credential.identityToken!,
      })
      if (error) Alert.alert('Error', error.message)
    } catch (e: any) {
      if (e.code !== 'ERR_REQUEST_CANCELED') Alert.alert('Error', e.message)
    }
  }

  async function handleGoogle() {
    const { error } = await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { redirectTo: 'tidy://auth/callback' },
    })
    if (error) Alert.alert('Error', error.message)
  }

  async function handleEmail() {
    setLoading(true)
    const fn = isSignUp
      ? supabase.auth.signUp({ email, password })
      : supabase.auth.signInWithPassword({ email, password })
    const { error } = await fn
    setLoading(false)
    if (error) Alert.alert('Error', error.message)
  }

  return (
    <View style={styles.container}>
      <LinearGradient {...PrimaryGradient} style={styles.logoBox}>
        <Text style={styles.star}>✦</Text>
      </LinearGradient>
      <Text style={styles.wordmark}>tidy</Text>
      <Text style={styles.tagline}>Chores, handled.</Text>

      {Platform.OS === 'ios' && (
        <AppleAuthentication.AppleAuthenticationButton
          buttonType={AppleAuthentication.AppleAuthenticationButtonType.SIGN_IN}
          buttonStyle={AppleAuthentication.AppleAuthenticationButtonStyle.BLACK}
          cornerRadius={Radii.pill}
          style={styles.appleBtn}
          onPress={handleApple}
        />
      )}

      <TouchableOpacity style={styles.googleBtn} onPress={handleGoogle}>
        <Text style={styles.googleText}>Continue with Google</Text>
      </TouchableOpacity>

      <View style={styles.divider}>
        <View style={styles.dividerLine} />
        <Text style={styles.dividerText}>or</Text>
        <View style={styles.dividerLine} />
      </View>

        style={styles.input}
        placeholder="Email"
        value={email}
        onChangeText={setEmail}
        autoCapitalize="none"
        keyboardType="email-address"
      />
      <TextInput
        style={styles.input}
        placeholder="Password"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
      />
      <TouchableOpacity style={styles.emailBtn} onPress={handleEmail} disabled={loading}>
        <Text style={styles.emailBtnText}>{isSignUp ? 'Create account' : 'Sign in'}</Text>

}
```

```typescript
      .on('postgres_changes', { event: '*', schema: 'public', table: 'house_members' }, fetchHouse)
      .subscribe()

    return () => { supabase.removeChannel(channel) }
  }, [userId])

  async function fetchHouse() {
    setLoading(true)
    const { data: membership } = await supabase
      .from('house_members')
      .select('house_id')
      .eq('user_id', userId)
      .single()

    if (!membership) { setHouse(null); setMembers([]); setLoading(false); return }

    const [{ data: houseData }, { data: membersData }] = await Promise.all([
      supabase.from('houses').select('*').eq('id', membership.house_id).single(),
      supabase.from('house_members').select('*, profile:profiles(*)').eq('house_id', membership.house_id),
    ])

    setHouse(houseData)
    setMembers(membersData ?? [])
    setLoading(false)
  }

  async function createHouse(name: string): Promise<{ error: string | null }> {
    const code = generateCode()
    const { data, error } = await supabase.from('houses').insert({ name, created_by: userId, code }).select().single()
    if (error) return { error: error.message }
    const { error: memberError } = await supabase.from('house_members').insert({ house_id: data.id, user_id: userId })
    return { error: memberError?.message ?? null }
  }

  async function joinHouse(code: string): Promise<{ error: string | null }> {
    const { data: house, error: findError } = await supabase
      .from('houses').select('id').eq('code', code.toUpperCase()).single()
    if (findError || !house) return { error: 'House not found. Check your code and try again.' }
    const { error } = await supabase.from('house_members').insert({ house_id: house.id, user_id: userId })
    if (error?.code === '23505') return { error: "You're already in a house. Leave your current house first." }
    return { error: error?.message ?? null }
  }

  async function leaveHouse(): Promise<{ error: string | null }> {
    // Unassign this user's chores first
    if (house) {
      await supabase.from('chores')
        .update({ assigned_to: null })
        .eq('house_id', house.id)
        .eq('assigned_to', userId)
    }
    const { error } = await supabase.from('house_members').delete().eq('user_id', userId)
    return { error: error?.message ?? null }
  }

  return { house, members, loading, createHouse, joinHouse, leaveHouse, refetch: fetchHouse }
}

function generateCode(): string {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'
  return 'TIDY-' + Array.from({ length: 4 }, () => chars[Math.floor(Math.random() * chars.length)]).join('')
}
```

```typescript
import { Chore, ChoreCompletion } from '@/types/database'

// Returns true if a completion record exists for this chore on this date
export function isChoreCompleteForDate(
  choreId: string,
  forDate: string,
  completions: ChoreCompletion[]
): boolean {
  return completions.some(c => c.chore_id === choreId && c.for_date === forDate)
}

// Returns the ISO date string (YYYY-MM-DD) for Monday of the week containing the given date
export function getMondayOfWeek(dateStr: string): string {
  const date = new Date(dateStr)
  const day = date.getUTCDay() // 0=Sun, 1=Mon...
  const diff = day === 0 ? -6 : 1 - day
  const monday = new Date(date)
  monday.setUTCDate(date.getUTCDate() + diff)
  return monday.toISOString().slice(0, 10)
}

// Returns chores that should appear on a given day
export function getChoresForDay(chores: Chore[], forDate: string): Chore[] {
  const monday = getMondayOfWeek(forDate)
  return chores.filter(chore => {
    if (chore.frequency === 'daily') return true
    // Weekly with due_date: appears only on that day
    if (chore.due_date) return chore.due_date === forDate
    // Weekly without due_date: appears on Monday of the week
    return forDate === monday
  })
}

// Returns progress dot color for a set of assigned chores + completions on a given day
// Returns null if no assigned chores exist for that day (no dot shown)
export function getProgressDotColor(
  assignedChoresForDay: Chore[],
  completions: ChoreCompletion[],
  forDate?: string
): 'red' | 'orange' | 'yellow' | 'green' | null {
  if (assignedChoresForDay.length === 0) return null
  const date = forDate ?? new Date().toISOString().slice(0, 10)
  const doneCount = assignedChoresForDay.filter(c =>
    isChoreCompleteForDate(c.id, date, completions)
  ).length
  const ratio = doneCount / assignedChoresForDay.length
  if (ratio === 1) return 'green'
  if (ratio >= 0.5) return 'yellow'
  if (ratio >= 0.25) return 'orange'
  return 'red'
}

// Sort chores: incomplete first (by due_date asc, created_at asc), completed last
export function sortChores(
  chores: Chore[],
  completions: ChoreCompletion[],
  forDate: string
): Chore[] {
  return [...chores].sort((a, b) => {
    const aDone = isChoreCompleteForDate(a.id, forDate, completions)
    const bDone = isChoreCompleteForDate(b.id, forDate, completions)
    if (aDone !== bDone) return aDone ? 1 : -1
    if (a.due_date && b.due_date) return a.due_date.localeCompare(b.due_date)
    if (a.due_date) return -1
    if (b.due_date) return 1
    return a.created_at.localeCompare(b.created_at)
  })
}
```

```typescript
  const [completions, setCompletions] = useState<ChoreCompletion[]>([])
  const [loading, setLoading] = useState(true)

  const fetchAll = useCallback(async () => {

    setLoading(true)
    const [{ data: choreData }, { data: completionData }] = await Promise.all([
      supabase.from('chores').select('*, assigned_profile:profiles!assigned_to(*)').eq('house_id', houseId),
      supabase.from('chore_completions')
        .select('*')
        .in('chore_id', (await supabase.from('chores').select('id').eq('house_id', houseId)).data?.map(c => c.id) ?? [])
    ])
    setChores(choreData ?? [])
    setCompletions(completionData ?? [])
    setLoading(false)
  }, [houseId])

  useEffect(() => {
    fetchAll()
    if (!houseId) return
    const channel = supabase.channel(`chores:${houseId}`)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'chores', filter: `house_id=eq.${houseId}` }, fetchAll)
      .on('postgres_changes', { event: '*', schema: 'public', table: 'chore_completions' }, fetchAll)
      .subscribe()
    return () => { supabase.removeChannel(channel) }
  }, [houseId, fetchAll])

  // Returns chores for selected day, with completion status attached
  function getChoresForSelectedDay(filterUserId?: string): ChoreWithCompletion[] {
    let dayChores = getChoresForDay(chores, selectedDate)
    if (filterUserId !== undefined) {
      // Avatar selected: show only that person's assigned chores
      dayChores = dayChores.filter(c => c.assigned_to === filterUserId)
    }
    const sorted = sortChores(dayChores, completions, selectedDate)
    return sorted.map(chore => ({

    }))

  async function markDone(choreId: string, userId: string): Promise<void> {
    // Optimistic update
    const optimistic: ChoreCompletion = {
      id: `optimistic-${choreId}`,
      chore_id: choreId,
      completed_by: userId,
      completed_at: new Date().toISOString(),
      for_date: selectedDate,
    }
    setCompletions(prev => [...prev, optimistic])
    const { error } = await supabase.from('chore_completions').insert({
      chore_id: choreId,
      completed_by: userId,
      for_date: selectedDate,
    })
    if (error) {
      // Rollback optimistic update
      setCompletions(prev => prev.filter(c => c.id !== optimistic.id))
    }
  }

  async function undoMarkDone(choreId: string): Promise<void> {
    setCompletions(prev => prev.filter(c => !(c.chore_id === choreId && c.for_date === selectedDate)))
    await supabase.from('chore_completions')
      .delete()
      .eq('chore_id', choreId)
      .eq('for_date', selectedDate)
  }

  async function addChore(params: {
    name: string
    assignedTo: string | null
    dueDate: string | null
    notes: string | null
    frequency: 'daily' | 'weekly'
    houseId: string
    createdBy: string
  }): Promise<{ error: string | null }> {

      name: params.name,
      assigned_to: params.assignedTo,
      due_date: params.dueDate,
      notes: params.notes,
      frequency: params.frequency,
      house_id: params.houseId,
      created_by: params.createdBy,
```

```typescript
import * as Notifications from 'expo-notifications'
import * as Device from 'expo-device'
import { Platform } from 'react-native'
import { supabase } from '@/lib/supabase'

Notifications.setNotificationHandler({
  handleNotification: async () => ({
    shouldShowAlert: true,
    shouldPlaySound: true,
    shouldSetBadge: true,
  }),
})

export async function registerForPushNotifications(userId: string): Promise<string | null> {
  if (!Device.isDevice) return null
  if (Platform.OS === 'android') {
    await Notifications.setNotificationChannelAsync('default', {
      name: 'default',
      importance: Notifications.AndroidImportance.MAX,
    })
  }

  const { status: existingStatus } = await Notifications.getPermissionsAsync()
  let finalStatus = existingStatus
  if (existingStatus !== 'granted') {
    const { status } = await Notifications.requestPermissionsAsync()
    finalStatus = status
  }
  if (finalStatus !== 'granted') return null

  const token = (await Notifications.getExpoPushTokenAsync()).data
  // Store push token on the user's profile
  await supabase.from('profiles').update({ push_token: token }).eq('id', userId)
  return token
}
```

```typescript
// Called by pg_cron every hour — sends 8am notification per user timezone
import { serve } from 'https://deno.land/std@0.177.0/http/server.ts'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

serve(async () => {
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  const nowUtc = new Date()

  // Get all profiles where it is currently 8am local time (within this hour)
  const { data: profiles } = await supabase
    .from('profiles')
    .select('id, timezone, push_token')
    .not('push_token', 'is', null)

  const messages: object[] = []

  for (const profile of profiles ?? []) {
    const localHour = new Date(nowUtc.toLocaleString('en-US', { timeZone: profile.timezone })).getHours()
    if (localHour !== 8) continue

    const todayLocal = nowUtc.toLocaleDateString('en-CA', { timeZone: profile.timezone }) // YYYY-MM-DD

    // Find assigned chores due today with no completion
    const { data: chores } = await supabase
      .from('chores')
      .select('id, name')
      .eq('assigned_to', profile.id)
      .eq('due_date', todayLocal)

    for (const chore of chores ?? []) {
      const { count } = await supabase
        .from('chore_completions')
        .select('*', { count: 'exact', head: true })
        .eq('chore_id', chore.id)
        .eq('for_date', todayLocal)

      if ((count ?? 0) === 0) {
        messages.push({
          to: profile.push_token,
          sound: 'default',
          title: 'Chore due today 🧹',
          body: `Don't forget: ${chore.name}`,
          data: { choreId: chore.id },
        })
      }
    }
  }

  if (messages.length > 0) {
    await fetch('https://exp.host/--/api/v2/push/send', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(messages),
    })
  }

  return new Response(`sent ${messages.length}`, { status: 200 })
})
```
